205  This is required because UDNL will only seach succeeding IOPRP images for modules specified within IOPBTCONF. */
206 static int generateIOPBTCONF_img(void *output, void *ioprp)

210  u8 *ptr_in;
211  u8 *ptr_out;
212

214  if (strcmp(romdir[0].name, "RESET") || strcmp(romdir[1].name, "ROMDIR"))
215  return -1;
216 
217  // Now locate IOPBTCONF
218  for (offset = 0; romdir->name[0] != '\0'; romdir++, offset += fsize_rounded) {
219  fsize_rounded = (romdir->size + 15) & ~15;
220  /* Unofficial: Check for already-patched image */
221  if (!strcmp(romdir->name, "IOPBTCONF") || !strcmp(romdir->name, "XOPBTCONF")) {
222  romdir->name[0] = 'X'; // Change 'IOPBTCONF' to 'XOPBTCONF', so that UDNL will not find the original.
223  // Copy IOPBTCONF into the new image.
224  ptr_out = (u8 *)output;
225  ptr_in = (u8 *)ioprp;
226  memcpy(ptr_out, &iopbtconf_img_base, sizeof(iopbtconf_img_base));
227  memcpy(ptr_out + sizeof(iopbtconf_img_base), &ptr_in[offset], romdir->size);
228  ((romdir_t *)ptr_out)[3].size = romdir->size; // Update the size of IOPBTCONF within the generated IOPRP image.
229  return romdir->size + sizeof(iopbtconf_img_base);
