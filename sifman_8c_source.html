121 void sceSifInit()
122 {
123  u32 msflag;
124  int state;
125  USE_IOP_MMIO_HWPORT();
126  USE_SIF_MMIO_HWPORT();
127 
128  if ( g_sif_inited )
129  return;
130  iop_mmio_hwport->dmac2.dpcr2 |= 0x8800;
131  iop_mmio_hwport->dmac2.newch[2].chcr = 0;
132  iop_mmio_hwport->dmac2.newch[3].chcr = 0;
133  sceSifDma2Init();
134  if ( (iop_mmio_hwport->iop_sbus_ctrl[0] & 0x10) )
135  iop_mmio_hwport->iop_sbus_ctrl[0] |= 0x10;
136  iop_mmio_hwport->iop_sbus_ctrl[0] |= 1;
137  sif_dma_init();
138  msflag = 0;
139  while ( !(msflag & SIF_STAT_SIFINIT) )
140  {
141  CpuSuspendIntr(&state);
142  msflag = get_msflag();
143  CpuResumeIntr(state);
144  }
145  sceSifSetDChain();
146  sceSifSetSubAddr(0);
147  sif_mmio_hwport->smflag = SIF_STAT_SIFINIT;
148  g_sif_inited = 1;
149 }

198 static int sifman_interrupt_handler(void *userdata)
199 {
200  int i;
201  sifman_internals_t *smi;
202  USE_IOP_MMIO_HWPORT();
203  USE_SIF_MMIO_HWPORT();
204 
205  smi = (sifman_internals_t *)userdata;
206  // Unofficial: unconditionally call callback
207  smi->m_dma_intr_handler(smi->m_dma_intr_handler_userdata);
208  for ( i = 0; i < smi->m_sif_curbufcom[0]->m_count; i += 1 )
209  smi->m_sif_curbufcom[0]->m_info[i].m_func(smi->m_sif_curbufcom[0]->m_info[i].m_userdata);
210  smi->m_sif_curbufcom[0]->m_count = 0;
211  if ( !sceSifIsSending() && smi->m_dmatag_index > 0 )
212  {
213  iop_mmio_hwport->dmac2.newch[2].chcr = 0;
214  iop_mmio_hwport->dmac2.newch[2].tadr = (uiptr)(smi->m_sif_curbuf);
215  iop_mmio_hwport->dmac2.newch[2].bcr = (iop_mmio_hwport->dmac2.newch[2].bcr & 0xFFFF0000) | 0x20;
216  if ( !(sif_mmio_hwport->controlreg & 0x20) )
217  sif_mmio_hwport->controlreg = 0x20;
218  smi->m_dma_count += 1;
219  smi->m_dmatag_index = 0;
220  smi->m_sif_curbufcom[1] = &smi->m_sif_bufcom[(smi->m_sif_curbuf == &smi->m_sif_buf[0]) ? 1 : 0];
221  smi->m_sif_curbufcom[0] = &smi->m_sif_bufcom[(smi->m_sif_curbuf == &smi->m_sif_buf[0]) ? 0 : 1];
222  smi->m_sif_curbuf = &smi->m_sif_buf[(smi->m_sif_curbuf == &smi->m_sif_buf[0]) ? 1 : 0];
223  iop_mmio_hwport->dmac2.newch[2].chcr = 0x1000701;
224  }
225  return 1;
226 }

228 static void sif_dma_init(void)
229 {
230  int state;
231 
232  g_sifman_internals.m_sif_curbufcom[1] = &g_sifman_internals.m_sif_bufcom[0];
233  g_sifman_internals.m_sif_curbufcom[0] = &g_sifman_internals.m_sif_bufcom[1];
234  g_sifman_internals.m_sif_curbuf = &g_sifman_internals.m_sif_buf[0];
235  g_sifman_internals.m_dmatag_index = 0;
236  g_sifman_internals.m_sif_bufcom[0].m_count = 0;
237  g_sifman_internals.m_sif_bufcom[1].m_count = 0;
238  sceSifResetDmaIntrHandler();
239  CpuSuspendIntr(&state);
240  RegisterIntrHandler(IOP_IRQ_DMA_SIF0, 1, sifman_interrupt_handler, &g_sifman_internals);
241  EnableIntr(IOP_IRQ_DMA_SIF0);
242  CpuResumeIntr(state);
243 }

245 static void sif_dma_setup_tag(const SifDmaTransfer_t *dmat)
246 {
247  sif_info_t *si;
248  unsigned int size_rounded;
249 
250  si = &g_sifman_internals.m_sif_curbuf->m_info[g_sifman_internals.m_dmatag_index];
251  si->m_data = ((uiptr)dmat->src & 0xFFFFFF) | ((dmat->attr & SIF_DMA_INT_I) ? 0x40000000 : 0);
252  size_rounded = dmat->size + 3;
253  si->m_words = (size_rounded >> 2) & 0xFFFFFF;
254  si->m_count =
255  ((size_rounded >> 4) + !!(si->m_words & 3)) | 0x10000000 | ((dmat->attr & SIF_DMA_INT_O) ? 0x80000000 : 0);
256  si->m_addr = (uiptr)dmat->dest & 0x1FFFFFFF;
257  g_sifman_internals.m_dmatag_index += 1;
258 }

260 static int set_dma_inner(const SifDmaTransfer_t *dmat, int count, void (*completioncb)(void *userdata), void *data)
261 {
262  u8 dmatag_index;
263  u16 dma_count;
264  int i;
265  USE_IOP_MMIO_HWPORT();
266  USE_SIF_MMIO_HWPORT();
267 
268  dmatag_index = g_sifman_internals.m_dmatag_index;
269  if (
270  dmatag_index + count
271  > (int)(sizeof(g_sifman_internals.m_sif_curbuf->m_info) / sizeof(g_sifman_internals.m_sif_curbuf->m_info[0])) )
272  return 0;
273  dma_count = g_sifman_internals.m_dma_count;
274  if ( dmatag_index )
275  g_sifman_internals.m_sif_curbuf->m_info[dmatag_index - 1].m_data &= ~0x80000000;
276  for ( i = 0; i < count; i += 1 )
277  sif_dma_setup_tag(&dmat[i]);
278  g_sifman_internals.m_sif_curbuf->m_info[g_sifman_internals.m_dmatag_index - 1].m_data |= 0x80000000;
279  if ( completioncb )
280  {
281  g_sifman_internals.m_sif_curbufcom[1]->m_info[g_sifman_internals.m_sif_curbufcom[1]->m_count].m_func = completioncb;
282  g_sifman_internals.m_sif_curbufcom[1]->m_info[g_sifman_internals.m_sif_curbufcom[1]->m_count].m_userdata = data;
283  g_sifman_internals.m_sif_curbufcom[1]->m_count += 1;
284  }
285  if ( !sceSifIsSending() && !iop_mmio_hwport->dmac2.new_unusedch.madr && !(iop_mmio_hwport->dmac2.dicr2 & 0x4000000) )
286  {
287  iop_mmio_hwport->dmac2.newch[2].chcr = 0;
288  iop_mmio_hwport->dmac2.newch[2].tadr = (uiptr)(g_sifman_internals.m_sif_curbuf);
289  if ( !(sif_mmio_hwport->controlreg & 0x20) )
290  sif_mmio_hwport->controlreg = 0x20;
291  iop_mmio_hwport->dmac2.newch[2].bcr = (iop_mmio_hwport->dmac2.newch[2].bcr & 0xFFFF0000) | 0x20;
292  g_sifman_internals.m_dmatag_index = 0;
293  g_sifman_internals.m_dma_count += 1;
294  g_sifman_internals.m_sif_curbufcom[1] =
295  &g_sifman_internals.m_sif_bufcom[(g_sifman_internals.m_sif_curbuf == &g_sifman_internals.m_sif_buf[0]) ? 1 : 0];
296  g_sifman_internals.m_sif_curbufcom[0] =
297  &g_sifman_internals.m_sif_bufcom[(g_sifman_internals.m_sif_curbuf == &g_sifman_internals.m_sif_buf[0]) ? 0 : 1];
298  g_sifman_internals.m_sif_curbuf =
299  &g_sifman_internals.m_sif_buf[(g_sifman_internals.m_sif_curbuf == &g_sifman_internals.m_sif_buf[0]) ? 1 : 0];
300  iop_mmio_hwport->dmac2.newch[2].chcr = 0x1000701;
301  }
302  return (dma_count << 16) | (dmatag_index << 8) | (count & 0xFF);
303 }
