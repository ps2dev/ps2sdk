103 int _start(int ac, char **av)
104 {
105  const int *BootMode3;
106  unsigned int i;
107 
108  (void)ac;
109  (void)av;
110 
111  BootMode3 = QueryBootMode(3);
112  if ( BootMode3 && (BootMode3[1] & 1) )
113  {
114  printf(" No SIF service(sifcmd)\n");
115  return MODULE_NO_RESIDENT_END;
116  }
117  if ( BootMode3 && (BootMode3[1] & 2) )
118  {
119  printf(" No SIFCMD/RPC service\n");
120  return MODULE_NO_RESIDENT_END;
121  }
122  if ( !sceSifCheckInit() )
123  sceSifInit();
124  if ( RegisterLibraryEntries(&_exp_sifcmd) )
125  return MODULE_NO_RESIDENT_END;
126  g_sif_cmd_data_table.m_pckt_buffer = g_sif_pckt_buffer;
127  g_sif_cmd_data_table.m_sys_buffer = g_sif_sys_buffer;
128  sceSifSetSysCmdBuffer(
129  g_sif_cmd_data_table.m_sys_cmd_handler_handler,
130  sizeof(g_sif_cmd_data_table.m_sys_cmd_handler_handler) / sizeof(g_sif_cmd_data_table.m_sys_cmd_handler_handler[0]));
131  g_sif_cmd_data_table.m_sif_send_eebuf = 0;
132  sceSifSetCmdBuffer(NULL, 0);
133  g_sif_cmd_data_table.m_sregs_ptr = g_sif_cmd_data_table.m_soft_reg;
134  sceSifClearSif1CB();
135  for ( i = 0; i < (unsigned int)g_sif_cmd_data_table.m_nr_sys_handlers; i += 1 )
136  sceSifRemoveCmdHandler(0x80000000 + i);
137  for ( i = 0; i < sizeof(g_sif_cmd_data_table.m_soft_reg) / sizeof(g_sif_cmd_data_table.m_soft_reg[0]); i += 1 )
138  sceSifSetSreg(i, 0);
139  sceSifAddCmdHandler(SIF_CMD_CHANGE_SADDR, _change_addr, (void *)&g_sif_cmd_data_table);
140  sceSifAddCmdHandler(SIF_CMD_SET_SREG, _set_sreg, (void *)&g_sif_cmd_data_table);
141  sceSifAddCmdHandler(SIF_CMD_INIT_CMD, sif_sys_cmd_handler_init_from_ee, (void *)&g_sif_cmd_data_table);
142  g_sif_cmd_data_table.m_ef = GetSystemStatusFlag();
143  RegisterIntrHandler(IOP_IRQ_DMA_SIF1, 1, _sceSifCmdIntrHdlr, &g_sif_cmd_data_table);
144  EnableIntr(0x200 | IOP_IRQ_DMA_SIF1);
145  sceSifSetSubAddr((u32)g_sif_cmd_data_table.m_pckt_buffer);
146  return MODULE_RESIDENT_END;
147 }

199 static int _sceSifSendCmd(
200  int cid,
201  char flags,
202  SifCmdHeader_t *packet,
203  int packet_size,
204  void *src_extra,
205  void *dest_extra,
206  int size_extra,
207  void (*completion_cb)(void *userdata),
208  void *completion_cb_userdata)
209 {
210  int dmatc;
211  unsigned int trid;
212  SifDmaTransfer_t dmat[2];
213  int state;
214 
215  if ( (unsigned int)(packet_size - 16) >= 0x61 )
216  return 0;
217  dmatc = 0;
218  packet->dest = (size_extra > 0) ? dest_extra : NULL;
219  packet->dsize = (size_extra > 0) ? size_extra : 0;
220  dmat[dmatc].dest = dest_extra;
221  dmat[dmatc].size = size_extra;
222  dmat[dmatc].attr = 0;
223  dmat[dmatc].src = src_extra;
224  dmatc += !!(size_extra > 0);
225  packet->psize = packet_size;
226  packet->cid = cid;
227  dmat[dmatc].src = packet;
228  dmat[dmatc].attr = SIF_DMA_INT_O;
229  dmat[dmatc].size = packet_size;
230  dmat[dmatc].dest = (void *)(g_sif_cmd_data_table.m_sif_send_eebuf);
231  if ( (flags & 1) )
232  return (flags & 8) ? sceSifSetDmaIntr(dmat, dmatc + 1, completion_cb, completion_cb_userdata) :
233  (unsigned int)sceSifSetDma(dmat, dmatc + 1);
234  CpuSuspendIntr(&state);
235  trid = (flags & 8) ? sceSifSetDmaIntr(dmat, dmatc + 1, completion_cb, completion_cb_userdata) :
236  (unsigned int)sceSifSetDma(dmat, dmatc + 1);
237  CpuResumeIntr(state);
238  return trid;
239 }

312 static int _sceSifCmdIntrHdlr(void *userdata)
313 {
314  SifCmdHeader_t *pkt;
315  int size;
316  int i;
317  u32 tmpbuf[32];
318  sif_cmd_data_t *sci;
319 
320  sci = (sif_cmd_data_t *)userdata;
321  // Unofficial: unconditionally call callback
322  sci->m_sif_1_callback(sci->m_sif_1_callback_userdata);
323  pkt = (SifCmdHeader_t *)sci->m_pckt_buffer;
324  size = pkt->psize;
325  if ( !size )
326  {
327  sceSifSetDChain();
328  return 1;
329  }
330  pkt->psize = 0;
331  size += 3;
332  size += (size < 0) ? 3 : 0;
333  tmpbuf[2] = 0; // cid
334  for ( i = 0; i < (size / (int)sizeof(tmpbuf[0])); i += 1 )
335  tmpbuf[i] = ((u32 *)pkt)[i];
336  pkt = (SifCmdHeader_t *)tmpbuf;
337  sceSifSetDChain();
338  if ( ((int)(pkt->cid & 0x7FFFFFFF) < ((!(pkt->cid & 0x80000000)) ? sci->m_nr_usr_handlers : sci->m_nr_sys_handlers)) && ((!(pkt->cid & 0x80000000)) ? g_sif_cmd_data_table.m_usr_cmd_handlers[pkt->cid].handler : g_sif_cmd_data_table.m_sys_cmd_handlers[(pkt->cid & 0x7FFFFFFF)].handler) )
339  ((!(pkt->cid & 0x80000000)) ? g_sif_cmd_data_table.m_usr_cmd_handlers[pkt->cid].handler :
340  g_sif_cmd_data_table.m_sys_cmd_handlers[(pkt->cid & 0x7FFFFFFF)].handler)(
341  pkt,
342  (!(pkt->cid & 0x80000000)) ? g_sif_cmd_data_table.m_usr_cmd_handlers[pkt->cid].harg :
343  g_sif_cmd_data_table.m_sys_cmd_handlers[(pkt->cid & 0x7FFFFFFF)].harg);
344  return 1;
345 }
